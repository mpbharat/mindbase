# Brain — Implementation Plan 1: Core Backend

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deploy a fully working cloud-hosted brain — Neon DB, Cloudflare Worker (MCP + REST API), mind-cli session hooks, and Claude Code + Hermes skills — so every agent on every machine shares memory automatically.

**Architecture:** A stateless Cloudflare Worker routes all requests to Neon Postgres (pgvector for semantic search, Workers AI for embeddings). mind-cli is a tiny Node.js CLI invoked by Claude Code session hooks. Skills tell Claude and Hermes what to do at session start/end.

**Tech Stack:** TypeScript, Cloudflare Workers, Wrangler CLI, Neon Postgres + pgvector, `@neondatabase/serverless`, `@modelcontextprotocol/sdk`, Cloudflare Workers AI (`@cf/baai/bge-base-en-v1.5`), Node.js CLI (commander + node-fetch), Vitest for tests.

Create `db/schema.sql`:
```sql
-- Enable pgvector extension
CREATE EXTENSION IF NOT EXISTS vector;

-- Projects (defined first — other tables reference it)
CREATE TABLE IF NOT EXISTS projects (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name        TEXT NOT NULL,
  status      TEXT DEFAULT 'active' CHECK (status IN ('active', 'paused', 'archived')),
  description TEXT,
  metadata    JSONB DEFAULT '{}',
  created_at  TIMESTAMPTZ DEFAULT now(),
  updated_at  TIMESTAMPTZ DEFAULT now()
);

-- Registered agents
CREATE TABLE IF NOT EXISTS agents (
  id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name        TEXT NOT NULL UNIQUE,
  type        TEXT NOT NULL CHECK (type IN ('claude-code', 'hermes', 'claude-desktop', 'custom')),
  machine     TEXT NOT NULL,
  status      TEXT DEFAULT 'offline' CHECK (status IN ('online', 'idle', 'offline')),
  last_seen   TIMESTAMPTZ,
  created_at  TIMESTAMPTZ DEFAULT now()
);

-- Live agent state (one row per agent, upserted)
CREATE TABLE IF NOT EXISTS agent_states (
  id             UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  agent_id       UUID REFERENCES agents(id) ON DELETE CASCADE UNIQUE,
  previous_task  TEXT,
  current_task   TEXT,
  next_task      TEXT,
  mood           TEXT DEFAULT 'idle' CHECK (mood IN ('idle', 'thinking', 'working', 'waiting', 'done')),
  context        JSONB DEFAULT '{}',
  updated_at     TIMESTAMPTZ DEFAULT now()
);

-- Task history per agent
CREATE TABLE IF NOT EXISTS agent_tasks (
  id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  agent_id     UUID REFERENCES agents(id) ON DELETE CASCADE,
  project_id   UUID REFERENCES projects(id) ON DELETE SET NULL,
  title        TEXT NOT NULL,
  description  TEXT,
  status       TEXT DEFAULT 'queued' CHECK (status IN ('queued', 'in_progress', 'done', 'cancelled')),
  priority     INT DEFAULT 2 CHECK (priority IN (1, 2, 3)),
  output       TEXT,
  started_at   TIMESTAMPTZ,
  completed_at TIMESTAMPTZ,
  created_at   TIMESTAMPTZ DEFAULT now()
);

-- Semantic memory store
-- Embedding: Cloudflare Workers AI @cf/baai/bge-base-en-v1.5 (768 dimensions)
CREATE TABLE IF NOT EXISTS memories (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  content    TEXT NOT NULL,
  embedding  vector(768),
  tags       TEXT[] DEFAULT '{}',
  project_id UUID REFERENCES projects(id) ON DELETE SET NULL,
  agent_id   UUID REFERENCES agents(id) ON DELETE SET NULL,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX IF NOT EXISTS memories_embedding_idx
  ON memories USING hnsw (embedding vector_cosine_ops);

CREATE INDEX IF NOT EXISTS memories_tags_idx
  ON memories USING gin (tags);

-- Backlog items
CREATE TABLE IF NOT EXISTS backlog_items (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  project_id UUID REFERENCES projects(id) ON DELETE CASCADE,
  title      TEXT NOT NULL,
  status     TEXT DEFAULT 'todo' CHECK (status IN ('todo', 'in_progress', 'done')),
  priority   INT DEFAULT 2 CHECK (priority IN (1, 2, 3)),
  notes      TEXT,
  created_at TIMESTAMPTZ DEFAULT now(),
  updated_at TIMESTAMPTZ DEFAULT now()
);

-- Session log
CREATE TABLE IF NOT EXISTS sessions (
  id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  agent_id   UUID REFERENCES agents(id) ON DELETE SET NULL,
  summary    TEXT,
  started_at TIMESTAMPTZ DEFAULT now(),
  ended_at   TIMESTAMPTZ
);
```

```typescript
import { handleAgents } from './routes/agents';
import { handleMemories } from './routes/memories';
import { handleProjects } from './routes/projects';
import { handleSessions } from './routes/sessions';
import { handleContext } from './context';
import { handleMcp } from './mcp';

export interface Env {
  AI: Ai;
  DATABASE_URL: string;
  MIND_API_KEY: string;
}

function unauthorized(): Response {
  return new Response(JSON.stringify({ error: 'Unauthorized' }), {
    status: 401,
    headers: { 'Content-Type': 'application/json' },
  });
}

function notFound(): Response {
  return new Response(JSON.stringify({ error: 'Not found' }), {
    status: 404,
    headers: { 'Content-Type': 'application/json' },
  });
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    // Auth check (skip for health endpoint)
    const url = new URL(request.url);
    if (url.pathname !== '/health') {
      const key = request.headers.get('X-Brain-Key');
      if (!key || key !== env.MIND_API_KEY) return unauthorized();
    }

    const path = url.pathname;

    if (path === '/health') return new Response('ok');
    if (path === '/mcp' || path.startsWith('/mcp/')) return handleMcp(request, env);
    if (path === '/context') return handleContext(request, env);
    if (path.startsWith('/agents')) return handleAgents(request, env, url);
    if (path.startsWith('/memories')) return handleMemories(request, env, url);
    if (path.startsWith('/projects') || path.startsWith('/backlog')) return handleProjects(request, env, url);
    if (path.startsWith('/sessions')) return handleSessions(request, env, url);

    return notFound();
  },
};
```

- [ ] **Step 5.1: Implement agents.ts**

Replace `mind-worker/src/routes/agents.ts`:
```typescript
import { Env } from '../index';
import { getDb } from '../lib/db';

function json(data: unknown, status = 200): Response {
  return new Response(JSON.stringify(data), {
    status,
    headers: { 'Content-Type': 'application/json' },
  });
}

export async function handleAgents(request: Request, env: Env, url: URL): Promise<Response> {
  const sql = getDb(env);
  const path = url.pathname; // e.g. /agents, /agents/:id/heartbeat, /agents/:id/state
  const method = request.method;
  const parts = path.split('/').filter(Boolean); // ['agents', id?, action?]

  // GET /agents — list all agents with their current state
  if (method === 'GET' && parts.length === 1) {
    const rows = await sql`
      SELECT a.*, s.current_task, s.previous_task, s.next_task, s.mood, s.updated_at as state_updated_at
      FROM agents a
      LEFT JOIN agent_states s ON s.agent_id = a.id
      ORDER BY a.last_seen DESC NULLS LAST
    `;
    return json(rows);
  }

  // POST /agents — register agent
  if (method === 'POST' && parts.length === 1) {
    const body = await request.json() as { name: string; type: string; machine: string };
    const [agent] = await sql`
      INSERT INTO agents (name, type, machine, status, last_seen)
      VALUES (${body.name}, ${body.type}, ${body.machine}, 'online', now())
      ON CONFLICT (name) DO UPDATE SET status = 'online', last_seen = now(), machine = ${body.machine}
      RETURNING *
    `;
    // Upsert an empty state row
    await sql`
      INSERT INTO agent_states (agent_id) VALUES (${agent.id})
      ON CONFLICT (agent_id) DO NOTHING
    `;
    return json(agent, 201);
  }

  const id = parts[1];
  const action = parts[2];

  // POST /agents/:id/heartbeat — keep alive
  if (method === 'POST' && action === 'heartbeat') {
    await sql`
      UPDATE agents SET status = 'online', last_seen = now() WHERE id = ${id}
    `;
    return json({ ok: true });
  }

  // PUT /agents/:id/state — update current/next task and mood
  if (method === 'PUT' && action === 'state') {
    const body = await request.json() as {
      current_task?: string;
      next_task?: string;
      previous_task?: string;
      mood?: string;
      context?: Record<string, unknown>;
    };
    const [state] = await sql`
      UPDATE agent_states SET
        current_task  = COALESCE(${body.current_task ?? null}, current_task),
        next_task     = COALESCE(${body.next_task ?? null}, next_task),
        previous_task = COALESCE(${body.previous_task ?? null}, previous_task),
        mood          = COALESCE(${body.mood ?? null}, mood),
        context       = COALESCE(${body.context ? JSON.stringify(body.context) : null}::jsonb, context),
        updated_at    = now()
      WHERE agent_id = ${id}
      RETURNING *
    `;
    return json(state);
  }

  // POST /agents/:id/tasks — add task to queue
  if (method === 'POST' && action === 'tasks') {
    const body = await request.json() as { title: string; description?: string; priority?: number; project_id?: string };
    const [task] = await sql`
      INSERT INTO agent_tasks (agent_id, project_id, title, description, priority)
      VALUES (${id}, ${body.project_id ?? null}, ${body.title}, ${body.description ?? null}, ${body.priority ?? 2})
      RETURNING *
    `;
    return json(task, 201);
  }

  // GET /agents/:id/tasks — list task queue
  if (method === 'GET' && action === 'tasks') {
    const rows = await sql`
      SELECT * FROM agent_tasks
      WHERE agent_id = ${id}
      ORDER BY
        CASE status WHEN 'in_progress' THEN 0 WHEN 'queued' THEN 1 ELSE 2 END,
        priority ASC,
        created_at ASC
    `;
    return json(rows);
  }

  // PATCH /agents/:id/tasks/:taskId — update task
  if (method === 'PATCH' && parts.length === 4 && parts[2] === 'tasks') {
    const taskId = parts[3];
    const body = await request.json() as { status?: string; output?: string };
    const updates: string[] = [];
    if (body.status) updates.push(`status = '${body.status}'`);
    if (body.output) updates.push(`output = '${body.output}'`);
    if (body.status === 'in_progress') updates.push(`started_at = now()`);
    if (body.status === 'done' || body.status === 'cancelled') updates.push(`completed_at = now()`);

    const [task] = await sql`
      UPDATE agent_tasks SET
        status       = COALESCE(${body.status ?? null}, status),
        output       = COALESCE(${body.output ?? null}, output),
        started_at   = CASE WHEN ${body.status} = 'in_progress' THEN now() ELSE started_at END,
        completed_at = CASE WHEN ${body.status} IN ('done', 'cancelled') THEN now() ELSE completed_at END
      WHERE id = ${taskId} AND agent_id = ${id}
      RETURNING *
    `;
    return json(task);
  }

  return new Response(JSON.stringify({ error: 'Not found' }), { status: 404 });
}
```

```typescript
const BASE_URL = process.env.MIND_URL ?? 'https://mind.YOUR_DOMAIN.com';
const API_KEY = process.env.MIND_API_KEY ?? '';

interface ContextOptions {
  agent?: string;
}

export async function runContext(options: ContextOptions): Promise<void> {
  if (!API_KEY) {
    process.stderr.write('MIND_API_KEY not set — skipping brain context\n');
    return;
  }

  const agentName = options.agent;
  if (!agentName) {
    process.stderr.write('MIND_AGENT_NAME not set — skipping brain context\n');
    return;
  }

  try {
    // Register / ping the agent first
    const regRes = await fetch(`${BASE_URL}/agents`, {
      method: 'POST',
      headers: { 'X-Brain-Key': API_KEY, 'Content-Type': 'application/json' },
      body: JSON.stringify({
        name: agentName,
        type: detectAgentType(),
        machine: detectMachine(),
      }),
    });
    const agent = await regRes.json() as { id: string };

    // Fetch context (markdown format)
    const ctxRes = await fetch(`${BASE_URL}/context?agent_id=${agent.id}`, {
      headers: { 'X-Brain-Key': API_KEY },
    });
    const context = await ctxRes.text();

    // Print to stdout — Claude Code session hook captures this and injects it
    process.stdout.write(context + '\n');
  } catch (err) {
    // Never crash the session over a brain failure
    process.stderr.write(`mind-cli context error: ${err}\n`);
  }
}

function detectAgentType(): string {
  return process.env.BRAIN_AGENT_TYPE ?? 'claude-code';
}

function detectMachine(): string {
  return process.env.BRAIN_MACHINE ?? require('os').hostname();
}
```

- [ ] **Step 12.3: Implement save.ts**

Create `mind-cli/src/save.ts`:
```typescript
import * as readline from 'readline';

const BASE_URL = process.env.MIND_URL ?? 'https://mind.YOUR_DOMAIN.com';
const API_KEY = process.env.MIND_API_KEY ?? '';

interface SaveOptions {
  agent?: string;
}

export async function runSave(options: SaveOptions): Promise<void> {
  if (!API_KEY || !options.agent) {
    return;
  }

  try {
    // Get agent id
    const agentsRes = await fetch(`${BASE_URL}/agents`, {
      headers: { 'X-Brain-Key': API_KEY },
    });
    const agents = await agentsRes.json() as Array<{ id: string; name: string }>;
    const agent = agents.find(a => a.name === options.agent);
    if (!agent) return;

    // Prompt Claude to provide session summary via stdin (hook environment)
    // In hook context, Claude writes to stdout which mind-cli reads
    const summary = await readStdin();

    if (summary.trim()) {
      // End the session
      const sessionId = process.env.BRAIN_SESSION_ID;
      if (sessionId) {
        await fetch(`${BASE_URL}/sessions/${sessionId}`, {
          method: 'PATCH',
          headers: { 'X-Brain-Key': API_KEY, 'Content-Type': 'application/json' },
          body: JSON.stringify({ summary: summary.trim(), agent_id: agent.id }),
        });
      }

      // Set agent mood to idle
      await fetch(`${BASE_URL}/agents/${agent.id}/state`, {
        method: 'PUT',
        headers: { 'X-Brain-Key': API_KEY, 'Content-Type': 'application/json' },
        body: JSON.stringify({ mood: 'idle', previous_task: summary.trim().slice(0, 200) }),
      });
    }
  } catch (err) {
    process.stderr.write(`mind-cli save error: ${err}\n`);
  }
}

async function readStdin(): Promise<string> {
  return new Promise(resolve => {
    let data = '';
    if (process.stdin.isTTY) {
      resolve('');
      return;
    }
    process.stdin.on('data', chunk => { data += chunk; });
    process.stdin.on('end', () => resolve(data));
    setTimeout(() => resolve(data), 3000); // 3s timeout
  });
}
```
